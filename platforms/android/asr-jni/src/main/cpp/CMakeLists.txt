cmake_minimum_required(VERSION 3.22)
project(saycribe_jni C)
set(CORE_INC ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../core/include)
set(JNI_LIBS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI})
set(PREBUILT ${JNI_LIBS_DIR}/libsaycribe_core.so)
if(EXISTS ${PREBUILT})
  add_library(saycribe_jni SHARED saycribe_jni.c)
  target_include_directories(saycribe_jni PRIVATE ${CORE_INC})
  target_link_directories(saycribe_jni PRIVATE ${JNI_LIBS_DIR})
  target_link_options(saycribe_jni PRIVATE "-Wl,-rpath-link,${JNI_LIBS_DIR}")
  target_link_libraries(saycribe_jni saycribe_core log)
else()
  message(WARNING "libsaycribe_core.so missing (${ANDROID_ABI}), using stub")
  add_library(saycribe_jni SHARED saycribe_jni.c saycribe_asr_stub.c)
  target_compile_definitions(saycribe_jni PRIVATE SC_STUB=1)
  target_link_libraries(saycribe_jni log)
  target_include_directories(saycribe_jni PRIVATE ${CORE_INC})
endif()
